from microbit import *
import obloq

WIFI_SSID = "School_WiFi"
WIFI_PASSWORD = "12345678"
SERVER_IP = "192.168.1.100"
SERVER_PORT = 5000
UPLOAD_ROUTE = "/upload"
TEMP_THRESHOLD = 30

obloq.setup(WIFI_SSID, WIFI_PASSWORD)

while True:
    raw = pin1.read_analog()
    temp = round(raw / 10, 1)
    display.scroll(str(temp))

    if temp > TEMP_THRESHOLD:
        pin2.write_digital(1)
    else:
        pin2.write_digital(0)

    url = "http://" + SERVER_IP + ":" + str(SERVER_PORT) + UPLOAD_ROUTE
    url = url + "?id=1&val=" + str(temp)
    obloq.http_get(url)

    sleep(5000)

Flask 服务端完整代码

from flask import Flask, request, render_template
import sqlite3
from datetime import datetime

app = Flask(__name__)
TEMP_THRESHOLD = 30

@app.route('/upload', methods=['GET'])
def upload_data():
    temperature = request.args.get('val')
    sensor_id = request.args.get('id', 1)
    if temperature is None:
        return "缺少 val", 400

    temperature = float(temperature)
    alarm = 1 if temperature > TEMP_THRESHOLD else 0

    conn = sqlite3.connect('sensor.db')
    cursor = conn.cursor()
    cursor.execute(
        'INSERT INTO sensorlog (sensor_id, value, alarm, timestamp) VALUES (?, ?, ?, ?)',
        (sensor_id, temperature, alarm, datetime.now())
    )
    conn.commit()
    conn.close()

    command = "BUZZER_ON" if alarm else "BUZZER_OFF"
    return "success:" + command

@app.route('/', methods=['GET'])
def index():
    conn = sqlite3.connect('sensor.db')
    cursor = conn.cursor()
    cursor.execute('SELECT id, sensor_id, value, alarm, timestamp FROM sensorlog ORDER BY id DESC LIMIT 10')
    rows = cursor.fetchall()
    conn.close()
    return render_template('index.html', records=rows)

if __name__ == '__main__':
    app.run(host='192.168.1.100', port=5000)

运行排错：先看画布，再看代码

运行后如果数据不通，先核对画布上的实际连接，再判断代码中的引脚是否对应。代码和连线可以任选一边修改，但最终必须对应；浏览器也要写完整地址才能访问页面。

排错步骤

  1. 回到画布，确认传感器 DATA 和蜂鸣器 IO 的实际引脚。
  2. 检查 read_analog() 读取的是哪个引脚，write_digital() 控制的是哪个引脚。
  3. 只改一个变量：要么改代码引脚，要么改画布连线，改完重新运行验证。
  4. 在模拟浏览器地址栏写完整 URL，再观察页面和数据库记录。
